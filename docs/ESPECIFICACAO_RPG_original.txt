package com.rpgclicker

import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.lifecycle.Lifecycle
import androidx.lifecycle.LifecycleObserver
import androidx.lifecycle.OnLifecycleEvent

import com.rpgclicker.data.repository.GameRepository

import com.rpgclicker.presentation.screen.MainScreen
import com.rpgclicker.presentation.viewmodel.GameViewModel
import com.rpgclicker.ui.theme.RpgClicker

class GameViewModelFactory(
    private val repository: GameRepository,
    private val gameEngine: GameEngine
) : ViewModelProvider.Factory {
    override fun <T : androidx.lifecycle.ViewModel> create(modelClass: Class<T>): T {
        return GameViewModel(repository, gameEngine) as T
    }
}

class MainActivity : ComponentActivity() {
    private val viewModel: GameViewModel by lazy {
        val application = application as RpgClickerApplication
        val database = application.gameDatabase!!
        val repository = GameRepository(database)
        val gameEngine = GameEngine(repository)
        ViewModelProvider(this, GameViewModelFactory(repository, gameEngine)).get(GameViewModel::class.java)
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        lifecycle.addObserver(AppLifecycleObserver(viewModel))
        setContent {
            RpgClicker {
                MainScreen(viewModel = viewModel)
            }
        }
    }
}

class AppLifecycleObserver(private val viewModel: GameViewModel) : LifecycleObserver {
    @OnLifecycleEvent(Lifecycle.Event.ON_RESUME)
    fun onAppForeground() {
        viewModel.onAppForeground()
    }
}

package com.rpgclicker.data.local

import android.content.Context
import androidx.room.Database
import androidx.room.Room
import androidx.room.RoomDatabase
import androidx.room.TypeConverters

@Database(
    entities = [
        GameStateEntity::class,
        EnemyEntity::class,
        AreaEntity::class,
        UpgradeEntity::class,
        PrestigeEntity::class,
        GameSettingsEntity::class,
        StatisticsEntity::class
    ],
    version = 1,
    exportSchema = false
)
@TypeConverters(Converters::class)
abstract class GameDatabase : RoomDatabase() {
    abstract fun gameDao(): GameDao

    companion object {
        @Volatile private var INSTANCE: GameDatabase? = null

        fun getDatabase(context: Context): GameDatabase {
            return INSTANCE ?: synchronized(this) {
                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    GameDatabase::class.java,
                    "rpg_clicker_db"
                ).fallbackToDestructiveMigration()
                    .build()
                INSTANCE = instance
                instance
            }
        }
    }
}

package com.rpgclicker.data.local

import androidx.room.Dao
import androidx.room.Insert
import androidx.room.OnConflictStrategy
import androidx.room.Query
import androidx.room.Update
import kotlinx.coroutines.flow.Flow

@Dao
interface GameDao {
    // Game State
    @Query("SELECT * FROM game_state WHERE id = 1")
    fun getGameState(): Flow<GameStateEntity>

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertGameState(entity: GameStateEntity)

    @Update
    suspend fun updateGameState(entity: GameStateEntity)

    // Enemies
    @Query("SELECT * FROM enemies WHERE id = :id")
    suspend fun getEnemy(id: String): EnemyEntity?

    @Query("SELECT * FROM enemies WHERE areaId = :areaId")
    suspend fun getEnemiesByArea(areaId: String): List<EnemyEntity>

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertEnemies(enemies: List<EnemyEntity>)

    @Update
    suspend fun updateEnemy(entity: EnemyEntity)

    // Areas
    @Query("SELECT * FROM areas WHERE id = :id")
    suspend fun getArea(id: String): AreaEntity?

    @Query("SELECT * FROM areas ORDER BY orderIndex")
    suspend fun getAllAreas(): List<AreaEntity>

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertAreas(areas: List<AreaEntity>)

    // Upgrades
    @Query("SELECT * FROM upgrades WHERE id = :id")
    suspend fun getUpgrade(id: String): UpgradeEntity?

    @Query("SELECT * FROM upgrades")
    suspend fun getAllUpgrades(): List<UpgradeEntity>

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertUpgrades(upgrades: List<UpgradeEntity>)

    // Prestige
    @Query("SELECT * FROM prestige WHERE id = 1")
    suspend fun getPrestige(): PrestigeEntity?

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertPrestige(entity: PrestigeEntity)

    @Update
    suspend fun updatePrestige(entity: PrestigeEntity)

    // Settings
    @Query("SELECT * FROM game_settings WHERE id = 1")
    suspend fun getSettings(): GameSettingsEntity?

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertSettings(entity: GameSettingsEntity)

    // Statistics
    @Query("SELECT * FROM statistics WHERE id = 1")
    suspend fun getStatistics(): StatisticsEntity?

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertStatistics(entity: StatisticsEntity)

    @Update
    suspend fun updateStatistics(entity: StatisticsEntity)
}

package com.rpgclicker.data.repository

import com.rpgclicker.data.local.GameDatabase
import com.rpgclicker.domain.model.*
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.map
import kotlinx.serialization.json.Json
import kotlinx.serialization.decodeFromString
import kotlinx.serialization.encodeToString
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import kotlinx.datetime.Clock
import kotlinx.datetime.Instant

class GameRepository(
    private val database: GameDatabase
) {
    private val dao = database.gameDao()
    private val json = kotlinx.serialization.json.Json { ignoreUnknownKeys = true }

    // Game State
    fun observeGameState(): Flow<GameState> = database.gameDao().getGameState()
        .map { entity -> if (entity == null) GameState() else deserializeGameState(entity) }

    suspend fun saveGameState(state: GameState) {
        val entity = serializeGameState(state)
        database.gameDao().insertGameState(entity)
    }

    suspend fun loadGameState(): GameState? = database.gameDao().getGameState().firstOrNull()?.let { deserializeGameState(it) }

    // Enemies
    suspend fun getEnemy(id: String): Enemy? = database.gameDao().getEnemy(id)?.toDomain()
    suspend fun getEnemiesByArea(areaId: String): List<Enemy> =
        database.gameDao().getEnemiesByArea(areaId).map { it.toDomain() }
    suspend fun saveEnemies(enemies: List<Enemy>) {
        database.gameDao().insertEnemies(enemies.map { it.toEntity() })
    }

    // Areas
    suspend fun getArea(id: String): Area? = database.gameDao().getArea(id)?.toDomain()
    suspend fun getAllAreas(): List<Area> =
        database.gameDao().getAllAreas().map { it.toDomain() }
    suspend fun saveAreas(areas: List<Area>) {
        database.gameDao().insertAreas(areas.map { it.toEntity() })
    }

    // Upgrades
    suspend fun getUpgrade(id: String): Upgrade? = database.gameDao().getUpgrade(id)?.toDomain()
    suspend fun getAllUpgrades(): List<Upgrade> =
        database.gameDao().getAllUpgrades().map { it.toDomain() }

    }

        database.gameDao().getAllUpgrades().map { it.map { it.toDomain() } }

    // Prestige

    // Settings

    }

    // Statistics

        database.gameDao().getStatistics()?.toDomain() ?: GameStatistics()
    suspend fun saveStatistics(stats: GameStatistics) {

    }

    // Serialization helpers
    private fun serializeGameState(state: GameState): GameStateEntity {
        return GameStateEntity(
            heroJson = json.encodeToString(state.hero),
            currentEnemyJson = state.currentEnemy?.let { json.encodeToString(it) },
            currentAreaId = state.currentAreaId,

    private fun deserializeGameState(entity: GameStateEntity): GameState {

package com.rpgclicker.domain.usecase

import com.rpgclicker.data.repository.GameRepository
import com.rpgclicker.domain.model.*
import kotlinx.coroutines.flow.Flow

class GameEngine(
    private val repository: GameRepository,
    private val mainDispatcher: CoroutineDispatcher = Dispatchers.Main,
    private val defaultDispatcher: CoroutineDispatcher = Dispatchers.Default
) {
    private var gameLoopJob: Job? = null

    fun observeGameState(): Flow<GameState> = repository.observeGameState()

    suspend fun onTapAttack(): AttackResult {
        val state = repository.observeGameState().first()
        val enemy = state.currentEnemy ?: return AttackResult(false, "Nenhum inimigo", 0, 0, false, false, false, false)

        val clickPower = state.hero.clickPower
        val critChance = state.hero.critChance
        val critDamage = state.hero.critDamageMultiplier

        val isCrit = kotlin.random.Random.nextDouble() < critChance
        val damage = if (isCrit) (clickPower * critDamage).toInt() else clickPower.toInt()

        val defeated = enemy.takeDamage(damage)

        val goldEarned = if (defeated) enemy.goldReward else 0
        val expEarned = if (defeated) enemy.expReward else 0

        var newState = state
        var newEnemy = enemy

        if (defeated) {
            val enemiesInArea = repository.getEnemiesByArea(state.currentAreaId)
            val currentIndex = enemiesInArea.indexOfFirst { it.id == enemy.id }
            if (currentIndex + 1 < enemiesInArea.size) {
                newEnemy = enemiesInArea[currentIndex + 1].copy(currentHealth = enemiesInArea[currentIndex + 1].maxHealth)
            } else {
                newEnemy = null
            }
        }

        newState = state.copy(
            currentEnemy = newEnemy,
            hero = state.hero.copy(
                gold = state.hero.gold + goldEarned,
                experience = state.hero.experience + expEarned
            ),
            statistics = state.statistics.copy(
                totalClicks = state.statistics.totalClicks + 1,
                totalGoldEarned = state.statistics.totalGoldEarned + goldEarned
            )
        )

        repository.saveGameState(newState)

        return AttackResult(
            success = true,
            message = if (defeated) "Inimigo derrotado!" else "Ataque!",
            damage = damage,
            goldEarned = goldEarned,
            expEarned = expEarned,
            enemyDefeated = defeated,
            isCritical = isCrit,
            enemy = newEnemy
        )
    }

    suspend fun purchaseUpgrade(upgradeId: String): PurchaseResult {
        val state = repository.observeGameState().first()
        val upgrades = repository.getAllUpgrades()
        val upgrade = upgrades.firstOrNull { it.id == upgradeId } ?: return PurchaseResult(false, "Upgrade nao encontrado", 0)

        val currentLevel = state.purchasedUpgrades[upgradeId] ?: 0
        if (currentLevel >= upgrade.maxLevel) return PurchaseResult(false, "Nivel maximo", currentLevel)

        val cost = upgrade.currentCost(currentLevel)
        if (state.hero.gold < cost) return PurchaseResult(false, "Ouro insuficiente", currentLevel)

        // Verificar prerequisito
        if (upgrade.prerequisiteId != null) {
            val prereqLevel = state.purchasedUpgrades[upgrade.prerequisiteId] ?: 0
            if (prereqLevel == 0) return PurchaseResult(false, "Pre-requisito nao atendido", currentLevel)
        }

        val newGold = state.hero.gold - cost
        val newPurchased = state.purchasedUpgrades + (upgradeId to (currentLevel + 1))

        var newState = state.copy(
            hero = state.hero.copy(gold = newGold),
            purchasedUpgrades = newPurchased,
            statistics = state.statistics.copy(totalGoldSpent = state.statistics.totalGoldSpent + cost)
        )

        // Aplicar efeitos do upgrade
        when (upgrade.type) {
            Upgrade.UpgradeType.CLICK_POWER -> {
                newState = newState.copy(
                    hero = newState.hero.copy(
                        clickPower = newState.hero.clickPower + upgrade.currentValue(currentLevel + 1).toInt()
                    )
                )
            }
            Upgrade.UpgradeType.AUTO_DPS -> {
                newState = newState.copy(
                    hero = newState.hero.copy(
                        autoDps = newState.hero.autoDps + upgrade.currentValue(currentLevel + 1)
                    )
                )
            }
            Upgrade.UpgradeType.GOLD_MULTIPLIER -> {
                // Implementar multiplicador de ouro
            }
            Upgrade.UpgradeType.HEALTH -> {
                newState = newState.copy(
                    hero = newState.hero.copy(
                        maxHealth = newState.hero.maxHealth + upgrade.currentValue(currentLevel + 1).toInt(),
                        currentHealth = newState.hero.currentHealth + upgrade.currentValue(currentLevel + 1).toInt()
                    )
                )
            }
            Upgrade.UpgradeType.CRIT_CHANCE -> {
                newState = newState.copy(
                    hero = newState.hero.copy(
                        critChance = newState.hero.critChance + upgrade.currentValue(currentLevel + 1)
                    )
                )
            }
            Upgrade.UpgradeType.CRIT_DAMAGE -> {
                newState = newState.copy(
                    hero = newState.hero.copy(
                        critDamageMultiplier = newState.hero.critDamageMultiplier + upgrade.currentValue(currentLevel + 1).toInt()
                    )
                )
            }
        }

        repository.saveGameState(newState)
        return PurchaseResult(true, "Upgrade comprado", currentLevel + 1)
    }

    suspend fun doPrestige(): PrestigeResult {
        val state = repository.observeGameState().first()
        val crystals = state.hero.gold / 1000 // Simplificado

        val newState = state.copy(
            hero = Hero(),
            currentEnemy = null,
            currentAreaId = "area_01",
            unlockedAreas = setOf("area_01"),
            purchasedUpgrades = emptyMap(),
            prestige = state.prestige.copy(
                crystals = state.prestige.crystals + crystals,
                totalCrystalsEarned = state.prestige.totalCrystalsEarned + crystals,
                globalMultiplier = state.prestige.globalMultiplier + 1
            ),
            statistics = state.statistics.copy(timesPrestiged = state.statistics.timesPrestiged + 1)
        )

        repository.saveGameState(newState)
        return PrestigeResult(success = true, crystalsEarned = crystals, newCrystals = newState.prestige.crystals)
    }

    fun calculateOfflineProgress(lastSaved: kotlinx.datetime.Instant): OfflineProgress {
        val now = Clock.System.now()
        val minutesElapsed = (now.toEpochMilliseconds() - lastSaved.toEpochMilliseconds()) / (1000 * 60)
        if (minutesElapsed <= 0) return OfflineProgress(0, 0)

        val state = repository.observeGameState().first()
        val goldEarned = (state.hero.autoDps * minutesElapsed).toLong()
        val expEarned = (minutesElapsed * 10).toInt()

        return OfflineProgress(goldEarned, expEarned)
    }

    suspend fun applyOfflineProgress() {
        val state = repository.observeGameState().first()
        val progress = calculateOfflineProgress(state.lastSaved)
        if (progress.gold > 0 || progress.exp > 0) {
            val newState = state.copy(
                hero = state.hero.copy(
                    gold = state.hero.gold + progress.gold,
                    experience = state.hero.experience + progress.exp
                ),
                lastSaved = Clock.System.now()
            )
            repository.saveGameState(newState)
        }
    }

    fun startAutoBattle() {
        gameLoopJob?.cancel()
        gameLoopJob = kotlinx.coroutines.launch(defaultDispatcher) {
            while (isActive) {
                withContext(mainDispatcher) {
                    onTapAttack()
                }
                kotlinx.coroutines.delay(1000)
            }
        }
    }

    fun stopAutoBattle() {
        gameLoopJob?.cancel()
        gameLoopJob = null
    }
}

class GameViewModel(
    private val repository: GameRepository,
    private val gameEngine: GameEngine
) : androidx.lifecycle.ViewModel() {

    private val _gameState = MutableStateFlow<GameState>(GameState())
    val gameState: StateFlow<GameState> = _gameState.asStateFlow()

    val upgrades: StateFlow<List<Upgrade>> = repository.getAllUpgradesFlow()
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(), emptyList())

    private val _lastAttackResult = MutableStateFlow<AttackResult?>(null)
    val lastAttackResult: StateFlow<AttackResult?> = _lastAttackResult.asStateFlow()

    private val _offlineProgress = MutableStateFlow<OfflineProgress?>(null)
    val offlineProgress: StateFlow<OfflineProgress?> = _offlineProgress.asStateFlow()

    private val _autoBattleEnabled = MutableStateFlow<Boolean>(false)
    val autoBattleEnabled: StateFlow<Boolean> = _autoBattleEnabled.asStateFlow()

    init {
        loadGame()
    }

    private fun loadGame() {
        viewModelScope.launch {
            repository.observeGameState().collect { state ->
                _gameState.value = state
            }
        }
    }

    fun onTapAttack() {
        viewModelScope.launch {
            val result = gameEngine.onTapAttack()
            _lastAttackResult.value = result
        }
    }

    fun purchaseUpgrade(upgradeId: String) {
        viewModelScope.launch {
            val result = gameEngine.purchaseUpgrade(upgradeId)
            // Atualizar estado apos compra
        }
    }

    fun doPrestige() {
        viewModelScope.launch {
            val result = gameEngine.doPrestige()
        }
    }

    fun toggleAutoBattle() {
        _autoBattleEnabled.value = !_autoBattleEnabled.value
        viewModelScope.launch {
            if (_autoBattleEnabled.value) {
                gameEngine.startAutoBattle()
            } else {
                gameEngine.stopAutoBattle()
            }
        }
    }

    fun onAppForeground() {
        viewModelScope.launch {
            val state = repository.observeGameState().first()
            val progress = gameEngine.calculateOfflineProgress(state.lastSaved)
            if (progress.gold > 0 || progress.exp > 0) {
                _offlineProgress.value = progress
                gameEngine.applyOfflineProgress()
            }
        }
    }

    fun dismissOfflineProgress() {
        _offlineProgress.value = null
    }

    fun dismissAttackResult() {
        _lastAttackResult.value = null
    }

    override fun onCleared() {
        super.onCleared()
        gameEngine.stopAutoBattle()
    }
}

@Composable
fun EnemyCard(
    modifier: Modifier = Modifier,
    enemy: Enemy,
    onAttack: () -> Unit
) {
    RpgClicker {
        val healthPercent = enemy.healthPercent
        val animatedHealthState = animateFloatAsState(healthPercent, animationSpec = spring(dampingRatio = 0.8f))

        Box(
            modifier = modifier
                .fillMaxWidth()
                .height(300.dp)
                .padding(16.dp)
                .clip(RoundedCornerShape(16.dp))
        ) {
            Box(
                modifier = Modifier
                    .fillMaxWidth()
                    .height(200.dp)
                    .padding(16.dp)
                    .background(Color(0xFFFFFFFF).copy(alpha = 0.1f))
                    .clip(RoundedCornerShape(12.dp))
            ) {
                EnemySprite(
                    spriteIndex = enemy.spriteIndex,
                    modifier = Modifier.fillMaxSize()
                )
            }

            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(start = 16.dp, end = 16.dp, bottom = 16.dp),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween
                ) {
                    Text(
                        text = enemy.name,
                        style = MaterialTheme.typography.titleMedium,
                        fontWeight = FontWeight.Bold
                    )
                    Text(
                        text = "${formatNumber(enemy.currentHealth.toLong())} / ${formatNumber(enemy.maxHealth.toLong())}",
                        style = MaterialTheme.typography.bodyMedium,
                        color = MaterialTheme.colorScheme.onSurfaceVariant
                    )
                }

                androidx.compose.foundation.layout.Spacer(modifier = Modifier.padding(top = 8.dp))

                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(12.dp)
                        .background(Color(0xFF1E1E2E))
                        .clip(RoundedCornerShape(6.dp))
                ) {
                    Box(
                        modifier = Modifier
                            .fillMaxWidth(animatedHealthState.value)
                            .fillMaxHeight()
                            .background(
                                if (animatedHealthState.value > 0.5f) Color(0xFF00D4AA)
                                else if (animatedHealthState.value > 0.25f) Color(0xFFFFD700)
                                else Color(0xFFFF4757)
                            )
                            .clip(RoundedCornerShape(6.dp))
                    )
                }

                Row(
                    modifier = Modifier.fillMaxWidth().padding(top = 8.dp),
                    horizontalArrangement = Arrangement.SpaceAround
                ) {
                    RewardItem(
                        iconLabel = "coin",
                        iconColor = Color.White,
                        iconBackground = Color(0xFFFFD700),
                        label = "Ouro",
                        value = formatNumber(enemy.goldReward.toLong()),
                        color = Color(0xFFFFD700)
                    )
                    RewardItem(
                        iconLabel = "star",
                        iconColor = Color.White,
                        iconBackground = Color(0xFF9B59B6),
                        label = "XP",
                        value = formatNumber(enemy.expReward.toLong()),
                        color = Color(0xFF9B59B6)
                    )
                }
            }
        }
    }
}

@Composable
private fun RewardItem(
    iconLabel: String,
    iconColor: Color,
    iconBackground: Color,
    label: String,
    value: String,
    color: Color,
    canAfford: Boolean = true
) {
    Column(horizontalAlignment = Alignment.CenterHorizontally) {
        GameIcon(
            name = iconLabel,
            modifier = Modifier.size(20.dp),
            colorFilter = if (canAfford) null else androidx.compose.ui.graphics.ColorFilter.tint(Color(0xFF8B949E))
        )
        androidx.compose.foundation.layout.Spacer(modifier = Modifier.padding(top = 4.dp))
        Text(
            text = value,
            style = MaterialTheme.typography.labelMedium,
            fontWeight = FontWeight.Bold,
            color = color
        )
        Text(
            text = label,
            style = MaterialTheme.typography.labelSmall,
            color = MaterialTheme.colorScheme.onSurfaceVariant
        )
    }
}

package com.rpgclicker.presentation.util

import androidx.compose.ui.unit.dp
import com.rpgclicker.R

object GameIcons {
    val byName: Map<String, Int> = mapOf(
        "fist" to R.drawable.ic_fist,
        "glove" to R.drawable.ic_glove,
        "gauntlet" to R.drawable.ic_gauntlet,
        "goblin" to R.drawable.ic_goblin,
        "wolf" to R.drawable.ic_wolf,
        "skeleton" to R.drawable.ic_skeleton,
        "coin" to R.drawable.ic_coin,
        "bag" to R.drawable.ic_bag,
        "heart" to R.drawable.ic_heart,
        "eye" to R.drawable.ic_eye,
        "dagger" to R.drawable.ic_dagger,
        "star" to R.drawable.ic_star,
        "hero" to R.drawable.sprite_hero,
        "enemy" to R.drawable.sprite_enemy
    )

    fun get(name: String?): Int = byName[name ?: "enemy"] ?: R.drawable.sprite_enemy
}

@Composable
fun GameIcon(
    name: String?,
    modifier: Modifier = Modifier,
    size: androidx.compose.ui.unit.Dp = 24.dp,
    colorFilter: androidx.compose.ui.graphics.ColorFilter? = null
) {
    val drawableId = GameIcons.get(name)
    val painter = painterResource(id = drawableId)
    androidx.compose.foundation.Image(
        painter = painter,
        contentDescription = null,
        modifier = modifier.size(size),
        colorFilter = colorFilter
    )
}

@Composable
fun EnemySprite(
    spriteIndex: Int,
    modifier: Modifier = Modifier,
    size: androidx.compose.ui.unit.Dp = 64.dp
) {
    val drawableId = when (spriteIndex) {
        0 -> R.drawable.enemy_slime
        1 -> R.drawable.enemy_goblin
        2 -> R.drawable.enemy_wolf
        3 -> R.drawable.enemy_bandit
        4 -> R.drawable.enemy_orc
        5 -> R.drawable.enemy_skeleton
        6 -> R.drawable.enemy_troll
        7 -> R.drawable.enemy_dragon_whelp
        8 -> R.drawable.enemy_dark_knight
        9 -> R.drawable.enemy_ancient_dragon
        else -> R.drawable.sprite_enemy
    }
    val painter = painterResource(id = drawableId)
    androidx.compose.foundation.Image(
        painter = painter,
        contentDescription = null,
        modifier = modifier.size(size)
    )
}

package com.rpgclicker.presentation.util

fun formatNumber(value: Long): String {
    return when {
        value >= 1_000_000_000 -> "${(value / 1_000_000_000.0).format("%.1f")}B"
        value >= 1_000_000 -> "${(value / 1_000_000.0).format("%.1f")}M"
        value >= 1_000 -> "${(value / 1_000.0).format("%.1f")}k"
        else -> value.toString()
    }
}

fun Double.format(decimals: Int): String {
    return java.text.DecimalFormat("0." + "0".repeat(decimals)).format(this)
}
